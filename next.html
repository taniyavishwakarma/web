// Framework Architecture Example
// Playwright Page Object pattern with self-healing fallback selectors
import { Page, Locator, expect } from '@playwright/test';

export class TransactionPage {
  readonly page: Page;
  readonly confirmButton: Locator;

  constructor(page: Page) {
    this.page = page;
    // Primary accessibility selector with backup data-testid attribute
    this.confirmButton = page.getByRole('button', { name: /confirm order/i })
      .or(page.locator('[data-testid="btn-confirm-transaction"]'));
  }

  async submitTransaction(): Promise<void> {
    await expect(this.confirmButton).toBeVisible({ timeout: 5000 });
    await this.confirmButton.click();
    await this.page.waitForResponse(resp => 
      resp.url().includes('/api/v1/orders') && resp.status() === 200
    );
  }
}
// Automated ETL & API verification against database record
import { test, expect } from '@playwright/test';
import { executeQuery } from '../helpers/dbClient';

test('Verify order creation syncs accurately with data store', async ({ request }) => {
  const payload = { userId: 'usr_8844', amount: 1250, currency: 'USD' };
  
  const response = await request.post('/api/v1/checkout', { data: payload });
  expect(response.status()).toBe(201);
  const { transactionId } = await response.json();

  // Validate state consistency in backend table
  const records = await executeQuery(
    `SELECT status, amount FROM transactions WHERE id = $1`, [transactionId]
  );
  
  expect(records[0].status).toBe('COMPLETED');
  expect(Number(records[0].amount)).toBe(payload.amount);
});
# Automated Performance Quality Gate for GitHub Actions / Azure Pipelines
name: Performance Verification Gate

on: [pull_request]

jobs:
  performance-gate:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Run Distributed JMeter Suite
        run: |
          jmeter -n -t tests/load-profile.jmx -l results.jtl -e -o ./report
      - name: Enforce p95 & Error Budgets
        run: |
          python3 scripts/verify_thresholds.py \
            --file results.jtl \
            --max-p95-ms 800 \
            --max-error-rate 0.01 \
            --min-throughput 250
